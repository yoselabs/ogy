```yaml
persona:
  member:         {nature: human, means: "a coworking member who books rooms through the app"}
  community-host: {nature: human, means: "approves bookings of bigger rooms; reads member notes"}
  cleaner:        {nature: human, means: "gets the room ready after a meeting"}

  book-room:      {states: [requested, held, confirmed, done]}
  see-usage:      {states: [seen]}

term:
  late-cancel:    {means: "a cancellation made after the late-cancel window starts"}
  hold:           {means: "the room reserved for the member while payment is awaited"}
  bigger-room:    {means: "a room whose capacity is above the approval threshold"}

set:
  plan:
    members:
      standard: {}   # ?open: plans other than premium not stated
      premium:  {}
    ?open:
      question: "Which plans exist besides premium, and does each pay per booking?"
      assumed: "exactly two plans: premium (credits) and everyone else (Stripe)"
      asks: persona/member

external:
  stripe:   {owns_facts: [payment], events: [payment-confirmed], pin: ?open}
  sms:      {owns_facts: [sms-delivery], pin: ?open}
  kisi:     {owns_facts: [door-code-registered], pin: ?open}

value:
  late-cancel-window:
    is: ?open {question: "How long before the start counts as late?", assumed: "unknown, no number assumed"}
    unit: duration
  late-fee:
    is: ?open {question: "How much is the late fee (fixed, percent of price)?", assumed: "unknown"}
    unit: money
  approval-capacity:
    is: ?open {question: "From how many seats is a room 'bigger'?", assumed: "unknown"}
    unit: number
  payment-hold-duration:
    is: ?open {question: "How long is a room held awaiting payment?", assumed: "unknown"}
    unit: duration
  credits-per-booking:
    is: ?open {question: "How many credits does a booking cost?", assumed: "unknown"}

entity:
  room:
    attrs:
      name:     {type: text}
      capacity: {type: number}
    key: [name]
  member:
    attrs:
      plan:    {type: set/plan}
      credits: {type: counter}   # ?open: starts at 0 per spec; real balance source unknown
    ?open:
      question: "How do credits get added to a member's balance, and what is the starting balance?"
      assumed: "credits are a counter changed only by booking confirmation and refunds (if any)"
      asks: persona/member
  booking:
    attrs:
      member:    {type: entity/member}
      room:      {type: entity/room}
      slot_start: {type: date}
      slot_end:   {type: date}
      attendees: {type: number}
      note:      {type: text, optional: true}
      fee:       {type: money, optional: true}
    lifecycle:
      states: [pending-payment, confirmed, cancelled, finished]
      initial: pending-payment
      terminal: [cancelled, finished]
      transitions:
        confirm: {from: pending-payment, to: confirmed}
        cancel:  {from: [pending-payment, confirmed], to: cancelled}
        finish:  {from: confirmed, to: finished}
      ?open:
        question: "Where does a booking waiting for host approval sit, and does a premium booking skip pending-payment?"
        assumed: "approval is a step before the hold (no booking row exists until approved); premium starts in pending-payment and confirms at once"
        asks: persona/community-host

derived:
  needs-approval: {is: "booking.room.capacity > value/approval-capacity"}   # ?open: approval based on room capacity or attendees?
  room-free: {is: "count(booking[room = this.room and overlaps slot and state in [pending-payment, confirmed]]) = 0"}
  is-late: {is: "booking.slot_start - now < value/late-cancel-window"}

workflow:
  request-booking:
    goal: goal/book-room
    actors: [persona/member, system, persona/community-host, external/stripe]
    trigger: {actor: persona/member}
    inputs: {room: entity/room, slot_start: date, slot_end: date, attendees: number}
    start: check-room
    steps:
      check-room:
        actor: system
        cases:
          taken:   {order: 1, when: "not derived/room-free", then: end/room-taken}
          big:     {order: 2, when: "derived/needs-approval", then: step/host-approval}
          premium: {order: 3, when: "member.plan = premium", then: step/hold-premium}
        otherwise: {then: step/hold-and-pay}
        causes: []
      host-approval:
        actor: persona/community-host
        choices:
          approve: {label: "Approve", then: step/after-approval}
          decline: {label: "Decline", then: end/declined}
        ?timeout: {after: ?open, then: end/declined}
        ?open:
          question: "What if the host never answers? Can the host give a reason on decline?"
          assumed: "no timeout; booking stays waiting"
          asks: persona/community-host
      after-approval:
        actor: system
        cases:
          premium: {order: 1, when: "member.plan = premium", then: step/hold-premium}
        otherwise: {then: step/hold-and-pay}
      hold-and-pay:
        actor: system
        then: workflow/pay-booking
        causes: [{create: booking}, {tell: persona/member, what: "payment request via Stripe"}]
      hold-premium:
        actor: system
        then: workflow/confirm-booking
        causes: [{create: booking}]
    ends:
      room-taken: {result: failure, means: "room not free for the slot"}
      declined:   {result: failure, means: "host declined or booking not approved"}
      handed-off: {result: success, means: "booking continues in payment or confirmation"}

  pay-booking:
    goal: goal/book-room
    actors: [system, external/stripe, persona/member]
    trigger: {called}
    inputs: {booking: entity/booking}
    start: await-payment
    steps:
      await-payment:
        actor: external/stripe
        choices:
          paid:   {label: "Stripe confirms payment", then: workflow/confirm-booking}
          failed: {label: "Payment failed", then: end/released}
        ?timeout: {after: value/payment-hold-duration, then: end/released}
        causes: []
    ends:
      released: {result: failure, means: "hold released, booking not confirmed"}
      ?open:
        question: "On failed or missing payment, is the booking cancelled and the room freed?"
        assumed: "booking goes to cancelled, no fee"
        asks: persona/member

  confirm-booking:
    goal: goal/book-room
    actors: [system, external/sms, external/kisi]
    trigger: {called}
    inputs: {booking: entity/booking}
    start: confirm
    steps:
      confirm:
        actor: system
        cases:
          premium-no-credits: {order: 1, when: "member.plan = premium and member.credits < value/credits-per-booking", then: end/no-credits}
        otherwise: {then: step/issue-code}
        causes:
          - {transition: booking.confirm}
          - {count: member.credits, op: dec}   # premium only; ?open how many (value/credits-per-booking)
        ?conflict: {then: end/no-credits}
      issue-code:
        actor: system
        then: step/send-code
        causes: [{tell: external/kisi, what: "door code for the slot"}]
      send-code:
        actor: system
        then: end/confirmed
        causes: [{tell: external/sms, what: "door code to the member"}]
    ends:
      confirmed:  {result: success, means: "booking confirmed, code sent and registered"}
      no-credits: {result: failure, means: "premium member lacks credits"}
    ?open:
      question: "What if SMS or Kisi fails/is down? What if premium credits are insufficient?"
      assumed: "no retry rules decided; premium with too few credits cannot confirm (ends no-credits)"
      asks: persona/member

    goal: goal/book-room
    actors: [persona/member, system]
    trigger: {actor: persona/member}
    inputs: {booking: entity/booking}
    start: decide-fee
    steps:
      decide-fee:
        actor: system
        cases:
          late: {order: 1, when: "derived/is-late", then: end/cancelled-late}
        otherwise: {then: end/cancelled}
        causes: [{transition: booking.cancel}]

      cancelled:      {result: success, means: "cancelled without a fee"}
      cancelled-late: {result: success, means: "cancelled, late fee charged", gives: late-fee}
    ?open:
      question: "Do premium members pay a late fee or lose credits? Are credits refunded on early cancel? Is a fee charged on a pending-payment booking? Can a booking be cancelled after it started? Is paid money refunded on early cancel? Is the door code revoked in Kisi on cancel?"
      assumed: "late fee as money charged through Stripe; no refunds, no credits restored, code not revoked"
      asks: persona/member

  prepare-room:
    goal: goal/book-room
    actors: [persona/cleaner, system]
    trigger: {event: transition/booking.finish}
    inputs: {booking: entity/booking}
    start: clean
    steps:
      clean:
        actor: persona/cleaner
        choices:
          ready: {label: "Room is ready", then: end/ready}
    ends:
      ready: {result: success, means: "room prepared for next use"}

  finish-booking:
    goal: goal/book-room
    actors: [system]
    trigger: {after: ?open, since: transition/booking.confirm}
    ?open:
      question: "What ends a booking: the slot end time? Does something mark finished, or does the host?"
      assumed: "system marks finished when slot_end passes (trigger is clock at slot_end)"
      asks: persona/community-host
    inputs: {booking: entity/booking}
    start: finish
    steps:
      finish: {actor: system, then: end/finished, causes: [{transition: booking.finish}]}
    ends:
      finished: {result: success, means: "booking finished"}
```

## Questions for the owner

- How long before the start counts as a late cancel (late-cancel window)?
- How is the late fee computed (fixed amount, share of price)?
- What makes a room "bigger": capacity above how many seats, or attendees above how many?
- Is approval keyed on the room's size or on the number of attendees booked?
- What is the booking state while waiting for host approval; is it one of the four states, or another?
- Is the room held while waiting for the host's approval?
- What if the host never answers, or declines; does the member get told and can the host give a reason?
- How long is the hold while waiting for payment, and what happens when it runs out (cancelled, freed)?
- What happens when Stripe payment fails or is refused?
- Do premium bookings skip payment, and at what state do they start; do premium members of bigger rooms still need approval?
- How many credits does a booking cost, and how do credits get into a member's balance?
- What if a premium member lacks enough credits at confirmation?
- Do premium members pay a late fee, lose credits, or get credits back on cancel?
- Is money refunded when a paid booking is cancelled early or late?
- Which plans exist besides premium?
- Is the door code revoked in Kisi on cancellation?
- What if SMS or Kisi fails or is unavailable at confirmation; retry, tell the host, tell the member?
- Does the door code have a validity window (only for the slot)?
- How many attendees can a room take; is exceeding capacity refused?
- What marks a booking finished: slot end time, or something a person does?
- Does the cleaner report back that the room is ready, and what happens if not done?
- Can a booking be cancelled after the meeting started or finished?
- What does usage history show (bookings, credits, fees), for which period, and may a member see others'?
- Who can read and who can edit the note, and until when; is there a reply from the host?
- Can a member change a booking (time, room, attendees) instead of cancel and rebook?
- Who may do what in the app: can the host or staff cancel or book for a member?
- Does a double-booking race (two members at once) resolve first-come?
- Are the Stripe, SMS provider and Kisi versions/accounts to pin (external pin)?
