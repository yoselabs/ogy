# product/meeting-rooms/book-room.yml
workflow:
  book-room:
    goal: goal/rent-rooms
    actors: [persona/member, persona/community-host, external/stripe, external/sms-provider, external/kisi]
    trigger: {actor: persona/member}
    inputs: {room: entity/room, slot_start: date, slot_end: date, attendees: number}
    start: step/check-free
    ends:
      unavailable: {result: failure, means: "the room is not free for the slot", gives: "tell the member"}
      declined: {result: failure, means: "the host declined the booking"}
      payment_failed: {result: failure, means: "payment did not complete in time or failed; the hold is released"}
      confirmed: {result: success, means: "booking confirmed, door code sent"}
    steps:
      check-free:
        actor: system
        cases:
          free: {order: 1, when: "derived/room_free", then: step/route-approval, causes: [{create: booking}]}
          otherwise: {order: 2, when: otherwise, then: end/unavailable, causes: [{tell: persona/member, what: "room not free"}]}
      route-approval:
        actor: system
        cases:
          big: {order: 1, when: "derived/needs_approval", then: step/host-approval}
          premium: {order: 2, when: "derived/is_premium", then: step/confirm}
          otherwise: {order: 3, when: otherwise, then: step/await-payment}
        ?open: {question: "Order for a big room and premium member, and is approval asked before or after the room is held? Where does the premium path go after approval?", assumed: "approval comes first; after approval, premium goes to confirm, others to await-payment"}
      host-approval:
        actor: persona/community-host
        choices:
          approve: {label: "Approve", then: step/after-approval, causes: [{write: booking.approved}]}
          decline: {label: "Decline", then: end/declined, causes: [{transition: booking.cancel}]}
        ?timeout: {after: ?open, then: end/declined}
        ?open: {question: "What if the host does not answer? Is there a deadline, and does the room stay held meanwhile?", assumed: "no timeout; the room stays held until the host answers"}
      after-approval:
        actor: system
        cases:
          premium: {order: 1, when: "derived/is_premium", then: step/confirm}
          otherwise: {order: 2, when: otherwise, then: step/await-payment}
      await-payment:
        actor: external/stripe
        choices:
          paid: {label: "payment_confirmed", then: step/confirm}
          failed: {label: "payment_failed", then: end/payment_failed, causes: [{transition: booking.cancel}]}
        ?timeout: {after: value/hold_duration, then: end/payment_failed}
        ?open: {question: "Who asks the member to pay, and how (in app, by SMS)? Is a payment link or a Stripe checkout used?", assumed: "the app shows a Stripe payment request"}
      confirm:
        actor: system
        then: step/send-code
        causes:
          - {transition: booking.pay}
          - {count: member.credits, op: dec, ?when: "member is premium"}
        ?open: {question: "A premium booking never passes through payment; does it go straight to confirmed? How many credits and what if the balance is too low?", assumed: "premium goes straight to confirmed, credits deducted at that moment, insufficient credits blocks (unspecified end)"}
      send-code:
        actor: system
        then: end/confirmed
        causes:
          - {write: booking.door_code}
          - {tell: external/kisi, what: "door code for the room and slot"}
          - {tell: persona/member, what: "door code by SMS"}
        ?open: {question: "How is the code made (random, per booking, per room)? When does it work (only during the slot)? What if SMS or Kisi fails?", assumed: "a code per booking, valid for the slot, failure of either is not handled"}

## Questions for the owner

- How long is a room held while waiting for payment, and what happens when it runs out?
- How far before the start does a cancellation become "late", and what is the late fee (fixed or a share)?
- How is the fee charged to a member who paid by credits (a premium member)? Does a premium member lose credits instead?
- If a member cancels in time, is money refunded and are credits returned? What about a cancelled booking after the code was sent: is the code revoked in Kisi?
- What counts as a "bigger room": a seat count? Is it a property of the room or of the number of attendees?
- Is approval asked before the hold and payment? Does the room stay held while waiting for the host, and what if the host never answers or declines?
- Does a big room booking by a premium member still need approval? In what order: approval, then credits?
- How many credits does a booking cost, and what if a premium member has too few?
- How do premium members receive credits (monthly grant, purchase)? What other plans exist besides premium?
- Do the credits get deducted only at confirmation, as stated, or is some credit held earlier to avoid double spending?
- Is a premium booking confirmed immediately, or does something else confirm it when no Stripe payment exists?
- What does the member see if Stripe payment fails or never arrives?
- Who creates the door code, what makes it valid (only the slot time?), and what if the SMS or Kisi call fails?
- What moves a booking to "finished": the end time passing, or a person? Is it automatic?
- Does the cleaner report that the room is ready, and does a room that is not clean block the next booking?
- Can the member edit the booking (change time, room, attendees) or only cancel?
- Who may cancel besides the member (host, owner)? Can the host cancel a booking?
- Who sees the note (host only?), until when can it be added, and is the host notified?
- What does usage history include (all states, fees, credits spent), and over what period?
- Is there a limit on how far ahead one can book, the slot length (whole hours?), or opening hours of the space?
- Are time slots and the "now" in one time zone of the space?
- Do the owner's states "pending payment" cover a booking waiting for host approval, or should it be a visible state?
- Are there any quality targets (e.g. the code must reach the member within some seconds)?
