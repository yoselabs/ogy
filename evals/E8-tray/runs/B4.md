# B4 capture: meeting rooms

```yaml
personas:
  member:   {nature: human}

  cleaner:  {nature: human}
  system:   {nature: agent}   # our app, acts on system steps

goals:
  book_room:
    states: [member holds a confirmed, paid-or-credited booking for a room and slot]
  see_usage:
    states: [member has seen their usage history]
  keep_rooms_ready:
    states: ?open {question: "is room readiness a goal of its own?", assumed: "yes, cleaner readies room after meeting"}

capabilities:
  book_a_room:      {does: "member picks room, slot, attendees; we hold, take payment or credits, confirm, send code", serves: book_room}
  approve_booking:  {does: "host approves bookings for bigger rooms", serves: book_room}
  cancel_booking:   {does: "member cancels; late cancel is charged a fee", serves: book_room}
  usage_history:    {does: "member sees their past usage", serves: see_usage}

  premium_plan: {means: "membership plan whose members pay with credits instead of money"}
  credit:       {means: "prepaid unit deducted for a premium booking"}
  door_code:    {means: "code that opens the room door for a booking"}
  late_cancel:  {means: "a cancellation after the cutoff"}
  bigger_room:  {means: ?open {question: "what makes a room bigger?", assumed: "a per-room flag requires_approval"}}

values:
  late_cancel_cutoff:  {is: ?open {question: "how long before start is a cancel late?", assumed: "unset"}, unit: ?open {question: "hours?", assumed: "hours"}}
  late_fee:            {is: ?open {question: "amount or share of price?", assumed: "unset"}, unit: ?open {question: "currency/percent?", assumed: "unset"}}
  payment_window:      {is: ?open {question: "how long is a room held awaiting payment?", assumed: "unset"}, unit: ?open {question: "minutes?", assumed: "minutes"}}
  approval_window:     {is: ?open {question: "how long may host take to approve?", assumed: "unset"}, unit: ?open {question: "hours?", assumed: "hours"}}
  big_room_capacity:   {is: ?open {question: "capacity above which approval is needed", assumed: "unset"}, unit: people}
  credits_per_booking: {is: ?open {question: "how many credits does a booking cost?", assumed: "unset"}, unit: credits}

sets:
  rooms:  {members: ?open {question: "which rooms exist?", assumed: "managed data, not fixed here"}}
  plans:  {members: [standard, premium]}   # standard name assumed; owner said only premium
  plans_note: ?open {question: "other plans besides premium?", assumed: "all non-premium pay by Stripe"}

entities:
  booking:
    key: booking_id
    attrs:
      room:        {type: ref room, ?max: na: "reference", writers: [create_booking], readers: [check_free, usage_history]}
      slot:        {type: time range, ?max: ?open {question: "longest slot?", assumed: "unbounded"}, writers: [create_booking], readers: [check_free, usage_history]}
      attendees:   {type: count, ?max: "room capacity", writers: [create_booking], readers: [check_capacity, check_big_room]}
      member:      {type: ref member, ?max: na: "reference", writers: [create_booking], readers: [usage_history, pay_path]}
      note:        {type: text, ?max: ?open {question: "note length limit?", assumed: "unbounded"}, writers: [add_note], readers: [host_view]}
      door_code:   {type: text, ?max: na: "generated", writers: [confirm_booking], readers: [send_code]}
      fee_charged: {type: money, ?max: na: "set by rule", writers: [cancel_late], readers: [usage_history]}
    lifecycle:
      states: [pending_payment, confirmed, cancelled, finished]
      initial: pending_payment
      terminal: [cancelled, finished]
      transitions:
        - {from: pending_payment, to: confirmed}
        - {from: pending_payment, to: cancelled}
        - {from: confirmed, to: cancelled}
        - {from: confirmed, to: finished}
      pending_approval: ?open {question: "where does a bigger-room booking wait for host approval: is it pending_payment, or a fifth state?", assumed: "extra state awaiting_approval before pending_payment, not in owner's four"}
      premium_initial: ?open {question: "premium booking skips payment: is it pending_payment too?", assumed: "pending_payment until credits are deducted"}
  member:
    key: member_id
    attrs:
      plan:     {type: plans, ?max: na: "set", writers: [?open {question: "who changes plan?", assumed: "outside this scope"}], readers: [pay_path]}
      credits:  {type: count, ?max: ?open {question: "cap on credits?", assumed: "none"}, writers: [deduct_credits], readers: [credits_enough]}
      phone:    {type: text, ?max: na: "phone number", writers: [?open {question: "where does phone come from?", assumed: "member profile"}], readers: [send_code]}
    lifecycle: ?open {question: "member lifecycle?", assumed: "out of scope"}

externals:
  stripe:
    owns_facts: [payment status]
    events: [payment_confirmed, payment_failed?]
    down: ?open {question: "if Stripe is down at hold time?", assumed: "booking stays pending, member told to retry"}
    slow: ?open {question: "payment arrives after hold expired?", assumed: "refund or reconfirm, undecided"}
    wrong_answer: ?open {question: "wrong amount confirmed?", assumed: "flag for host review"}
    answers_twice: "second confirmation ignored by booking_id"
    changes_behind_us: ?open {question: "refund/dispute by Stripe after confirm?", assumed: "no handling"}
  sms_provider:
    owns_facts: [message delivery status]
    events: [delivered?]
    down: ?open {question: "SMS down?", assumed: "retry, then host told"}
    slow: ?open {question: "late SMS acceptable?", assumed: "yes"}
    wrong_answer: ?open {question: "wrong number?", assumed: "none"}
    answers_twice: ?open {question: "duplicate text?", assumed: "harmless but unwanted"}
    changes_behind_us: na: "none stated"
  kisi:
    owns_facts: [door codes on the lock]
    events: ?open {question: "does Kisi report failures?", assumed: "synchronous ack only"}
    down: ?open {question: "Kisi down when code must be set?", assumed: "do not text code until Kisi has it"}
    slow: ?open {question: "slow Kisi?", assumed: "wait"}
    wrong_answer: ?open {question: "Kisi rejects code?", assumed: "booking not confirmed"}
    answers_twice: ?open {question: "duplicate code push?", assumed: "idempotent by booking"}
    changes_behind_us: ?open {question: "staff edit codes in Kisi directly?", assumed: "no"}

rules:
  big_room_needs_approval: {kind: guard, scope: create_booking, body: "attendees > big_room_capacity (or room flagged) requires host approval before payment"}
  premium_uses_credits:    {kind: guard, scope: pay_path, body: "premium plan skips Stripe and uses credits"}
  credits_deducted_at_confirm: {kind: adds, scope: confirm_booking, body: "deduct credits_per_booking when confirmed"}
  late_cancel_fee:         {kind: adds, scope: cancel_booking, body: "if now > start - late_cancel_cutoff, charge late_fee"}
  no_double_booking:       {kind: holds, scope: booking, body: "one non-cancelled booking per room and overlapping slot"}
  capacity:                {kind: holds, scope: booking, body: ?open {question: "attendees must not exceed room capacity?", assumed: "yes"}}

workflows:
  book_room_flow:
    goal: book_room
    actors: [member, host, system]
    trigger: member submits room, slot, attendees
    inputs:
      room:      {type: ref room}     # read by check_free
      slot:      {type: time range}   # read by check_free
      attendees: {type: count}        # read by big room guard
    start: check_free
    ends:
      confirmed:        {result: success, means: "booking confirmed, code sent"}
      room_unavailable: {result: failure, means: "room not free"}
      rejected:         {result: failure, means: "host declined"}
      payment_failed:   {result: failure, means: "payment not completed, hold released"}
      not_enough_credits: {result: failure, means: "premium member lacks credits"}
      # each end needs a message in every serving interface (see interfaces)
    steps:
      check_free:
        type: system
        cases:
          - {order: 1, when: "room not free for slot", then: end room_unavailable}
          - {order: 2, when: "room needs approval", then: host_approves}
          - {order: 3, when: "member is premium", then: confirm_booking}
        otherwise: {effect: "create booking (transition: pending_payment, tell: hold room)", then: pay}
      host_approves:
        type: actor
        actor: host
        choices:
          - {label: approve, then: route_after_approval}
          - {label: decline, then: end rejected}
        timeout: {after: approval_window, then: ?open {question: "what happens on no host answer?", assumed: "decline"}}
      route_after_approval:
        type: system
        cases:
          - {order: 1, when: "member is premium", then: confirm_booking}
        otherwise: {then: pay}
      pay:
        type: actor
        actor: member
        choices:
          - {label: "pay via Stripe", then: await_stripe}
          - {label: abandon, then: end payment_failed}
        timeout: {after: payment_window, then: end payment_failed}
      await_stripe:
        type: system
        cases:
          - {order: 1, when: "stripe confirms", then: confirm_booking}
          - {order: 2, when: "stripe fails", then: end payment_failed}
          - {order: 3, when: "no answer (unknown)", then: ?open {question: "no Stripe answer: wait, poll, or release?", assumed: "poll then release at payment_window"}}
        otherwise: {then: end payment_failed}
      confirm_booking:
        type: system
        cases:
          - {order: 1, when: "premium and credits < credits_per_booking", then: end not_enough_credits}
        otherwise:
          effects:
            - {transition: "booking pending_payment -> confirmed"}
            - {write: "door_code generated"}
            - {count: "member.credits -= credits_per_booking (premium only)"}
            - {tell: "push code to Kisi", rerun: checks, unknown_case: "no answer from Kisi", done_twice: "idempotent by booking_id", silently_not_done: ?open {question: "how do we notice Kisi never got the code?", assumed: "none"}}
            - {tell: "SMS door code to member", rerun: ?open {question: "resend on retry?", assumed: never}, unknown_case: "no answer from SMS", done_twice: "member gets two texts", silently_not_done: ?open {question: "how do we know member got the text?", assumed: "member can view code in app"}}
          then: end confirmed
  cancel_flow:
    goal: book_room
    actors: [member, system]
    trigger: member cancels a booking
    inputs:
      booking: {type: ref booking}   # read by late check
    start: check_state
    ends:
      cancelled:        {result: success, means: "booking cancelled, free or with late fee"}
      not_cancellable:  {result: failure, means: "booking already cancelled or finished"}
    steps:
      check_state:

        cases:

          - {order: 2, when: "now > start - late_cancel_cutoff", then: cancel_late}
        otherwise: {effect: "transition to cancelled; release room", then: end cancelled}
      cancel_late:
        type: system
        cases: []
        otherwise:
          effects:
            - {transition: "-> cancelled"}
            - {write: "fee_charged = late_fee"}
            - {tell: "charge late fee via Stripe", rerun: checks, unknown_case: "no answer", done_twice: "double charge unless idempotent", silently_not_done: ?open {question: "fee charge fails silently?", assumed: "recorded as unpaid"}}
          then: end cancelled
      cancel_refund: ?open {question: "refund of payment or credits on cancel?", assumed: "refund when on time, credits returned, undecided"}
  finish_flow:
    goal: keep_rooms_ready
    actors: [system, cleaner]
    trigger: meeting end time passes
    inputs: {booking: {type: ref booking}}
    start: mark_finished
    ends:
      room_ready:      {result: success, means: "room prepared"}
      room_not_ready:  {result: failure, means: ?open {question: "cleaner fails or never marks done?", assumed: "host alerted"}}
    steps:
      mark_finished:
        type: system
        cases: []
        otherwise: {effect: "transition confirmed -> finished; tell cleaner", then: prepare}
      prepare:
        type: actor
        actor: cleaner
        choices: [{label: "room ready", then: end room_ready}]
        timeout: {after: ?open {question: "how long may cleaning take?", assumed: "none"}, then: end room_not_ready}
  usage_flow:
    goal: see_usage
    actors: [member]
    trigger: member opens usage history
    inputs: {member: {type: ref member}}
    start: list
    ends:
      shown:  {result: success, means: "history displayed"}
      failed: {result: failure, means: "history could not be loaded"}
    steps:
      list:
        type: system
        cases: [{order: 1, when: "history unreadable", then: end failed}]
        otherwise: {then: end shown}
        what_counts: ?open {question: "which bookings in history, cancelled too, with fees and credits?", assumed: "all, with fee and credits"}
  note_flow:
    goal: book_room
    actors: [member, host]
    trigger: member adds a note
    inputs: {note: {type: text}, booking: {type: ref booking}}
    start: save
    ends:
      saved:  {result: success, means: "note stored for host"}
      refused: {result: failure, means: ?open {question: "can notes be added on cancelled/finished bookings?", assumed: "only pending or confirmed"}}
    steps:
      save:
        type: system
        cases: [{order: 1, when: "booking terminal", then: end refused}]
        otherwise: {effect: "write note; tell host?", then: end saved}
```

## Questions for the owner

- What makes a room "bigger": a capacity threshold or a per-room flag, and what is the number?
- Where does a booking wait for host approval: is it a fifth lifecycle state, or still pending payment?
- How long may the host take to approve, and what happens if the host never answers?
- If a bigger room is approved, is the room held while waiting for approval?
- How long is a room held while awaiting payment, and what happens when it expires?
- What if Stripe never answers, answers late (after hold expired), or confirms the wrong amount?
- Refunds or disputes on Stripe after confirmation: handled or not?
- How long before the start counts as a "late" cancel?
- How much is the late fee (amount or share), and how is it charged (Stripe, credits)?
- Are payments refunded on an on-time cancel, and are credits returned on cancel?
- Does a late fee apply to premium members, in money or in credits?
- How many credits does a booking cost, and what if a premium member lacks credits?
- Credits deducted on confirm: do premium bookings go through pending payment at all?
- Which plans exist besides premium; who changes plans?
- Can a cancel happen before payment, or after the meeting starts?
- What if Kisi is down or rejects the code; do we still confirm, and do we text the code?
- What if the SMS fails or the member's phone is missing or wrong; resend allowed?
- Can the code be seen in the app too, and does it expire after the meeting?
- Who does what when the cleaner never marks the room ready, and how is the cleaner told?
- Does the cleaner step block the next booking of the room?
- Who marks a booking finished: automatic at end time or by someone?
- What does usage history include (cancelled bookings, fees, credits) and over what period?
- Limits on notes: length, who can add, when (after cancel or finish), does the host get told?
- Does the host see all bookings or only those needing approval?
- Max attendees: must it not exceed room capacity? Minimum slot, maximum slot, slot granularity?
- What can a member see: which interface (app, web) and in which language?
- Can members modify a booking (change time, room, attendees), or only cancel?
- Is double booking of overlapping slots forbidden outright, and who wins a race for the same slot?
