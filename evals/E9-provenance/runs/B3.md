id: workflow/borrow
goal: goal/lend-tools
actors: [persona/member, persona/volunteer, external/stripe, system]
trigger: {actor: persona/member}
inputs: {tool: entity/tool}
start: step/browse
steps:
  browse:
    actor: persona/member
    choices:
      pick: {label: "Find a tool in the catalog", then: step/check-tool}
  check-tool:
    actor: system
    cases:
      free: {order: 1, when: "state/tool = available", then: step/pick-day}
    otherwise: {then: step/offer-waitlist}
  offer-waitlist:
    actor: persona/member
    choices:
      join:  {label: "Join the waitlist", then: workflow/join-waitlist, with: {tool: tool, member: member}}
      leave: {label: "Leave it", then: end/not-reserved}
  pick-day:
    actor: persona/member
    by: [rule/blocked-cannot-borrow, rule/power-needs-plus, rule/power-needs-safety-intro, rule/power-needs-adult, rule/loan-limit]
    choices:
      day: {label: "Reserve for this pickup day", then: step/counter, each: set/pickup-day,
            causes: [{create: entity/reservation}, {transition: entity/tool.reserve}]}
  counter:
    actor: persona/volunteer
    timeout: {after: value/pickup-grace, then: end/not-collected}
    choices:
      scan: {label: "Scan tool and member card", then: step/deposit}
  deposit:
    actor: system
    cases:
      power: {order: 1, when: "tool.is_power", then: step/charge-deposit}
    otherwise: {then: step/start-loan}
  charge-deposit:
    actor: external/stripe
    choices:
      charged:  {label: "Deposit charged", then: step/start-loan,
                 causes: [{write: entity/loan.deposit_status, from: "held"}]}
      declined: {label: "Deposit declined", then: end/refused-deposit, open: {decision: decision/d-payment-fails}}
  start-loan:
    actor: system
    then: end/borrowed
    causes: [{create: entity/loan}, {transition: entity/tool.start_loan}, {transition: entity/reservation.collect}]
